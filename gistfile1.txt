const { Client, GatewayIntentBits } = require('discord.js');
const { DisTube } = require('distube');
const { YtDlpPlugin } = require('@distube/yt-dlp');

const client = new Client({
  intents: [
    GatewayIntentBits.Guilds,
    GatewayIntentBits.GuildMessages,
    GatewayIntentBits.MessageContent,
    GatewayIntentBits.GuildVoiceStates
  ]
});

const TOKEN = 'MTQwNjc0MTI4MTAzOTcxNjQxNA.GKA22K.-jaXgYbSlSjpwKJQrs9Fis2ey1jjdNjDfPF8Y0'; // توكن البوت الخاص بك

const distube = new DisTube(client, {
  plugins: [new YtDlpPlugin()],
  leaveOnFinish: false,
  emitNewSongOnly: true
});

client.on('ready', () => {
  console.log(`Logged in as ${client.user.tag}!`);
});

client.on('messageCreate', async (message) => {
  if (message.author.bot) return;

  // اختصار التشغيل
  if (message.content === 'ش + p') {
    message.reply('اكتب اسم الأغنية فقط وسيتم تشغيلها!');
    return;
  }

  // اختصار الإيقاف
  if (message.content === 'س + s') {
    if (message.member.voice.channel) {
      distube.stop(message.member.voice.channel);
      message.reply('تم إيقاف الموسيقى!');
    } else {
      message.reply('يجب أن تكون في قناة صوتية!');
    }
    return;
  }

  // تشغيل أغنية عند كتابة اسمها فقط
  if (message.member.voice.channel) {
    // جرب تشغيل الأغنية
    try {
      await distube.play(message.member.voice.channel, message.content, {
        textChannel: message.channel,
        member: message.member
      });
      message.react('🎶');
    } catch (e) {
      message.reply('حدث خطأ أثناء محاولة تشغيل الأغنية.');
      console.error(e);
    }
  } else {
    if (message.content.trim().length > 1) {
      message.reply('انضم أولاً إلى قناة صوتية لأتمكن من تشغيل الأغنية!');
    }
  }
});

// رسائل من DisTube
distube
  .on('playSong', (queue, song) =>
    queue.textChannel.send(`🎵 الآن يتم تشغيل: \`${song.name}\``)
  )
  .on('addSong', (queue, song) =>
    queue.textChannel.send(`➕ تم إضافة: \`${song.name}\``)
  )
  .on('error', (channel, error) =>
    channel.send('⚠️ حدث خطأ: ' + error.toString())
  );

client.login(TOKEN);
