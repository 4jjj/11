@bot.command(name="play", help="شغل أغنية من يوتيوب")
async def play(ctx, *, search: str):
    if not ctx.author.voice:
        return await ctx.send("يجب أن تكون في قناة صوتية أولاً!")

    vc = ctx.voice_client

    if not vc:
        vc = await ctx.author.voice.channel.connect(cls=wavelink.Player)

    song = await wavelink.YouTubeTrack.search(search, return_first=True)
    await vc.play(song)
    await ctx.send(f"🎵 يتم تشغيل: {song.title}")

@bot.command(name="stop", help="إيقاف الموسيقى")
async def stop(ctx):
    if ctx.voice_client:
        await ctx.voice_client.disconnect()
        await ctx.send("تم إيقاف الموسيقى وفصل البوت.")
